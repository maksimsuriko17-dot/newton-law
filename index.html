import numpy as np import streamlit as st import plotly.graph_objects as go from plotly.subplots import make_subplots st.set_page_config(page_title="Второй закон Ньютона", page_icon="🚀", layout="wide") st.title("Второй закон Ньютона") st.subheader("Модель движения материальной точки в инерциальной системе отсчёта (ИСО)") st.write( "Изменяйте массу и равнодействующую силу. Модель покажет, " "как меняются ускорение, скорость и координата тела. " "Рассматривается движение вдоль одной прямой при постоянной силе." ) st.latex(r"\vec F_{\text{рез}} = m\vec a \qquad a = \frac{F_{\text{рез}}}{m}") with st.sidebar: st.header("Параметры модели") mass = st.slider("Масса тела, кг", min_value=0.5, max_value=20.0, value=5.0, step=0.5) force = st.slider("Равнодействующая сила, Н", min_value=-50, max_value=50, value=20, step=1) initial_velocity = st.slider("Начальная скорость, м/с", min_value=-20, max_value=20, value=0, step=1) initial_position = st.slider("Начальная координата, м", min_value=-20, max_value=20, value=0, step=1) duration = st.slider("Продолжительность моделирования, с", min_value=2, max_value=20, value=10, step=1) acceleration = force / mass # Время и кинематические величины для постоянного ускорения. time_values = np.linspace(0, duration, 301) position_values = initial_position + initial_velocity * time_values + 0.5 * acceleration * time_values**2 velocity_values = initial_velocity + acceleration * time_values selected_time = st.slider( "Выберите момент времени, с", min_value=0.0, max_value=float(duration), value=float(duration) / 2, step=0.1, ) selected_position = initial_position + initial_velocity * selected_time + 0.5 * acceleration * selected_time**2 selected_velocity = initial_velocity + acceleration * selected_time st.header("Результаты") col1, col2, col3, col4 = st.columns(4) col1.metric("Ускорение", f"{acceleration:.2f} м/с²") col2.metric("Скорость", f"{selected_velocity:.2f} м/с") col3.metric("Координата", f"{selected_position:.2f} м") col4.metric("Равнодействующая сила", f"{force} Н") st.caption("Положительное направление оси X направлено вправо. Отрицательная сила действует влево.") # Схема: положение материальной точки и направление равнодействующей силы. st.subheader("Движение материальной точки") fig_motion = go.Figure() all_x = list(position_values) + [selected_position] arrow_length = max(0.4, min(2.0, abs(force) / 20)) if force != 0: all_x.append(selected_position + np.sign(force) * arrow_length) left_bound = min(all_x) - 1.5 right_bound = max(all_x) + 1.5 if right_bound - left_bound < 4: middle = (right_bound + left_bound) / 2 left_bound, right_bound = middle - 2, middle + 2 fig_motion.add_trace( go.Scatter( x=[left_bound, right_bound], y=[0, 0], mode="lines", line=dict(width=3, color="#777777"), name="Ось X", hoverinfo="skip", ) ) fig_motion.add_trace( go.Scatter( x=[selected_position], y=[0], mode="markers+text", marker=dict(size=24, color="#E45756"), text=["Материальная точка"], textposition="top center", name="Тело", hovertemplate="Координата: %{x:.2f} м", ) ) if force > 0: fig_motion.add_annotation( x=selected_position + arrow_length, y=0.22, ax=selected_position, ay=0.22, xref="x", yref="y", axref="x", ayref="y", text="F", showarrow=True, arrowhead=3, arrowsize=1.4, arrowwidth=3, ) elif force < 0: fig_motion.add_annotation( x=selected_position - arrow_length, y=0.22, ax=selected_position, ay=0.22, xref="x", yref="y", axref="x", ayref="y", text="F", showarrow=True, arrowhead=3, arrowsize=1.4, arrowwidth=3, ) else: fig_motion.add_annotation( x=selected_position, y=0.35, text="F = 0", showarrow=False ) fig_motion.update_layout( xaxis_title="Координата, м", yaxis=dict(visible=False, range=[-0.5, 0.8]), xaxis=dict(range=[left_bound, right_bound], zeroline=False), height=300, margin=dict(l=30, r=30, t=30, b=50), showlegend=False, ) st.plotly_chart(fig_motion, use_container_width=True) # Графики координаты и скорости во времени. st.subheader("Графики движения") fig_graphs = make_subplots( rows=2, cols=1, shared_xaxes=True, vertical_spacing=0.15, subplot_titles=("Координата x(t)", "Скорость v(t)"), ) fig_graphs.add_trace( go.Scatter(x=time_values, y=position_values, mode="lines", name="x(t)"), row=1, col=1 ) fig_graphs.add_trace( go.Scatter(x=[selected_time], y=[selected_position], mode="markers", name="Выбранный момент"), row=1, col=1 ) fig_graphs.add_trace( go.Scatter(x=time_values, y=velocity_values, mode="lines", name="v(t)"), row=2, col=1 ) fig_graphs.add_trace( go.Scatter(x=[selected_time], y=[selected_velocity], mode="markers", name="Скорость в выбранный момент"), row=2, col=1 ) fig_graphs.update_xaxes(title_text="Время, с", row=2, col=1) fig_graphs.update_yaxes(title_text="Координата, м", row=1, col=1) fig_graphs.update_yaxes(title_text="Скорость, м/с", row=2, col=1) fig_graphs.update_layout(height=600, margin=dict(l=30, r=30, t=60, b=40), showlegend=False) st.plotly_chart(fig_graphs, use_container_width=True) st.subheader("Вывод") if force > 0: force_direction = "вправо" elif force < 0: force_direction = "влево" else: force_direction = "равнодействующая сила равна нулю" st.write( f"При массе {mass:.1f} кг и равнодействующей силе {force} Н " f"ускорение равно {acceleration:.2f} м/с². Направление ускорения: {force_direction}. " "При неизменной массе увеличение модуля силы увеличивает модуль ускорения; " "при неизменной силе увеличение массы уменьшает модуль ускорения." ) st.info("Модель описывает материальную точку в ИСО при постоянной равнодействующей силе; сопротивлением среды пренебрегаем.") streamlit numpy plotly
